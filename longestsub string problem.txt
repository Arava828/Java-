public class Main {
    static int longestSubstring(String s) {
        if (s == null || s.length() == 0) {
            return 0;
        }

        int max = 1;
        int current = 1;

        for (int i = 1; i < s.length(); i++) {
            if (s.charAt(i) != s.charAt(i - 1)) {
                current++;
            } else {
                current = 1;
            }
            max = Math.max(max, current);
        }
        return max;
    }

    public static void main(String[] args) {
        System.out.println(longestSubstring("xyxyzzz"));    // 5
        System.out.println(longestSubstring("vvuvuvuvuu")); // 8
    }
}
